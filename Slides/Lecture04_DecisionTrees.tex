\section{Why Go Beyond Linear Models?}
% =============================================================================

\sectionslide{Why Go Beyond Linear Models?}{%
  Real-world relationships are often nonlinear, involve interactions,
  and shift with context.}

% --- Slide: Linear model limitation ------------------------------------------
\begin{frame}{The Limits of Linear Models}
  \textbf{Three things a constant slope cannot represent:}
  \begin{itemize}
    \item \textbf{Interactions} --- the holiday-season advertising effect
          is not a constant slope
    \item \textbf{Threshold effects} --- spending may be flat until
          unemployment drops below 5\%, then jump
    \item \textbf{Asymmetry} --- extreme values distort coefficients
          equally in both directions
  \end{itemize}
  \begin{examplebox}{Retail Sales Example}
    Monthly sales depend on last year's same-month sales (trend),
    whether it is December (seasonality), and the unemployment rate
    (conditions). The December effect is \emph{larger} in a strong
    economy --- an interaction a linear model misses.
  \end{examplebox}
  \textbf{Decision trees capture nonlinearity and interactions automatically},
  without your having to specify them in advance.
\end{frame}

% --- Slide: Bias-variance tradeoff overview ----------------------------------
\begin{frame}{The Bias--Variance Tradeoff \parencite{James2021}}
  \begin{definitionbox}{Bias--Variance Decomposition}
    \[
      \text{Expected MSE} = \underbrace{\text{Bias}^2}_{\text{systematic error}}
                           + \underbrace{\text{Variance}}_{\text{sensitivity to data}}
                           + \underbrace{\sigma^2}_{\text{irreducible noise}}
    \]
  \end{definitionbox}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{High bias (underfitting):}
      \begin{itemize}\small
        \item Model is too simple
        \item Misses real patterns in the data
        \item Bad on both training AND test sets
        \item Example: linear model on a curved relationship
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{High variance (overfitting):}
      \begin{itemize}\small
        \item Model is too complex
        \item Memorizes training data noise
        \item Great on training, bad on test set
        \item Example: a very deep decision tree
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.15cm}
  \textbf{Goal:} find the complexity that balances the two --- which you can
  only see \emph{out-of-sample}.
\end{frame}

\section{How Decision Trees Work}
% =============================================================================

\sectionslide{How Decision Trees Work}{%
  Partition the feature space into rectangles; predict the mean in each region.}

% --- Slide: The tree metaphor ------------------------------------------------
\begin{frame}{What Is a Decision Tree?}
  A decision tree is a flowchart: ask yes/no questions about the features to
  arrive at a prediction.

  \vspace{-0.1cm}
  \begin{center}
    \begin{tikzpicture}[
      every node/.style={font=\small},
      internal/.style={rectangle, rounded corners=3pt,
        fill=unolightcrimson, draw=unocrimson, line width=0.8pt,
        align=center, inner sep=4pt,
        minimum width=2.4cm, minimum height=0.6cm},
      leaf/.style={rectangle, rounded corners=3pt,
        fill=unolightgreen, draw=unogreen, line width=0.8pt,
        align=center, inner sep=4pt,
        minimum width=1.8cm, minimum height=0.6cm},
    ]
      \node[internal] (root) at (3.6, 0) {$y_{t-12} \leq 38{,}000$?};
      \node[leaf] (l1) at (1.2, -1.35) {\footnotesize $\hat{y} = 28{,}400$};
      \node[internal] (r1) at (6.0, -1.35) {$y_{t-1} \leq 52{,}000$?};
      \node[leaf] (l2) at (4.6, -2.7) {\footnotesize $\hat{y} = 44{,}100$};
      \node[leaf] (r2) at (7.6, -2.7) {\footnotesize $\hat{y} = 61{,}300$};
      \draw[->, gray] (root) -- (l1) node[midway, above left, font=\tiny]{Yes};
      \draw[->, gray] (root) -- (r1) node[midway, above right, font=\tiny]{No};
      \draw[->, gray] (r1) -- (l2) node[midway, above left, font=\tiny]{Yes};
      \draw[->, gray] (r1) -- (r2) node[midway, above right, font=\tiny]{No};
    \end{tikzpicture}
  \end{center}
  \vspace{-0.25cm}
  {\small \textbf{Non-parametric} $\cdot$ \textbf{nonlinear} (captures
  thresholds) $\cdot$ \textbf{interpretable} $\cdot$ \textbf{scale-invariant}
  $\cdot$ handles lags, rolling statistics, and dummies together.}
  {\small A single \emph{deep} tree \textbf{overfits severely}. Ensembles are
  the fix, and are where the next two lectures go.}
\end{frame}

% --- Slide: CART algorithm ---------------------------------------------------
\begin{frame}{The CART Algorithm (Regression) \parencite{Breiman1984}}
  \textbf{CART} builds the tree greedily, top to bottom. At each node it
  searches over every feature $j$ and threshold $t$ for the split that most
  reduces within-region variance:
  \[
    \min_{j,\, t}
    \left[
      \sum_{i: x_{ij} \leq t} (y_i - \bar{y}_{L})^2
      +
      \sum_{i: x_{ij} > t} (y_i - \bar{y}_{R})^2
    \right]
  \]
  Start with all the data at the root; split until nodes get too small or a
  depth cap is reached. The \textbf{prediction} at a leaf is simply the mean
  of the observations that land there.
  At every node the algorithm is really asking: \emph{``what is the single
  yes/no question that best separates high-sales months from low-sales
  months?''} --- then repeating it within each group.
  \muted{\footnotesize For classification, replace variance with Gini impurity
  or entropy (information gain).}
\end{frame}

% --- Slide: greedy is not optimal --------------------------------------------
\begin{frame}{Greedy Is Not Optimal --- and That Matters}
  CART is \textbf{exhaustive} at each node: it really does try every feature and every
  threshold. But it is \textbf{greedy} across nodes --- it never looks ahead.
  \begin{center}
    {\small
    \begin{tabular}{lcc}
      \toprule
      \textbf{Split at the root} & \textbf{Variance removed now} & \textbf{\ldots plus what it enables} \\
      \midrule
      On price     & \textbf{40\%} & $+$ 12\% \quad (total 52\%) \\
      On promotion & 38\%          & $+$ 50\% \quad \textbf{(total 88\%)} \\
      \bottomrule
    \end{tabular}
    }
  \end{center}
  \vspace{-0.1cm}
  CART takes the 40\%. It has no way to see the 50\% coming, because it evaluates one split at
  a time and commits.

  \vspace{0.15cm}
  So a fitted tree is \textbf{not the optimal tree} --- it is the result of a sequence of
  locally optimal decisions, which is a different thing. Finding the genuinely optimal tree is
  computationally intractable for any realistic number of features, which is why nobody does it.

  \vspace{0.15cm}
  Two consequences worth carrying forward. A tree can \textbf{miss structure that is really
  there}. And a small change in the data can send the whole greedy sequence down a different
  path --- which is exactly the instability we turn to next.
\end{frame}

% --- Slide: Bias-variance in trees -------------------------------------------
\begin{frame}{Depth Controls the Bias--Variance Tradeoff}
  A \textbf{depth-1 stump} asks one question: high bias, low variance. A
  \textbf{full tree} with one observation per leaf memorizes the sample: zero
  bias, extreme variance.

  \vspace{0.15cm}
  \begin{center}
    {\small
    \begin{tabular}{llll}
      \toprule
      \textbf{Depth} & \textbf{Bias} & \textbf{Variance} & \textbf{Behavior} \\
      \midrule
      1 (stump)  & High     & Low      & One question; underfits \\
      5--10      & Balanced & Balanced & Usually the useful range \\
      Full tree  & Zero     & Extreme  & Memorizes the training sample \\
      \bottomrule
    \end{tabular}
    }
  \end{center}
  \vspace{0.1cm}
  \vspace{-0.2cm}
  \textbf{Why trees are unstable:} one changed observation can flip the split
  at the root, and every prediction below it changes with it.
  Tune \texttt{max\_depth} and \texttt{min\_samples\_leaf} by
  \texttt{TimeSeriesSplit} cross-validation.
\end{frame}

% --- Slide: pruning ----------------------------------------------------------
\begin{frame}{Two Ways to Stop a Tree Overfitting}
  \texttt{max\_depth} caps the tree in advance. Two other controls do the job better.
  \begin{center}
    {\small
    \begin{tabular}{p{3.2cm}p{9.2cm}}
      \toprule
      \texttt{min\_samples\_leaf} & \textbf{Stop early.} At the default of 1, a leaf may hold a
      single observation and ``predict'' that one value. Set it to 5--10 and every prediction is
      an average over a real group. \\[3pt]
      \texttt{ccp\_alpha} & \textbf{Prune after.} Grow the tree out fully, then charge a penalty
      $\alpha$ per leaf and cut back the branches that do not pay for themselves. \\
      \bottomrule
    \end{tabular}
    }
  \end{center}
  \vspace{-0.1cm}
  Cost-complexity pruning minimizes
  \vspace{-0.2cm}
  \[
    \underbrace{\textstyle\sum_{\text{leaves}} \text{SSE}}_{\text{fit}}
    \;+\; \underbrace{\alpha \cdot |T|}_{\text{complexity}},
    \qquad |T| = \text{number of leaves}
  \]
  \vspace{-0.25cm}

  \textbf{You have seen this before} --- it is the GAM smoothing penalty from Lecture 3, and it
  returns in Lecture 6 as Ridge and LASSO. Three model families, one idea, and $\alpha$ chosen
  the same way each time: by cross-validation.
\end{frame}

\section{Implementing Decision Trees in Python}
% =============================================================================

\sectionslide{Implementing Decision Trees in Python}{%
  One class, a few hyperparameters, interpretable results.}

% --- Slide: sklearn implementation -------------------------------------------
\begin{frame}{Decision Tree Regression in Python}
  \begin{examplebox}{sklearn Code Template}
    {\scriptsize
    \texttt{from sklearn.tree import DecisionTreeRegressor}\\
    \texttt{X = make\_lag\_features(df, lags=[1, 2, 3, 12]);\ \ y = df['sales']}\\
    \texttt{tree = DecisionTreeRegressor(max\_depth=5, min\_samples\_leaf=10, random\_state=42)}\\
    \texttt{tree.fit(X\_train, y\_train);\ \ y\_hat = tree.predict(X\_test)}
    }
  \end{examplebox}
  \vspace{-0.15cm}
  \begin{center}
    {\footnotesize
    \begin{tabular}{ll}
      \toprule
      \textbf{Parameter} & \textbf{Effect} \\
      \midrule
      \texttt{max\_depth}         & Cap tree depth \\
      \texttt{min\_samples\_leaf} & Minimum observations per leaf \\
      \texttt{max\_features}      & Feature subsampling \\
      \texttt{ccp\_alpha}         & Cost-complexity pruning \\
      \bottomrule
    \end{tabular}
    }
  \end{center}
  \vspace{-0.15cm}
  {\small Cross-validate with \texttt{TimeSeriesSplit}, never \texttt{KFold} ---
  \textbf{random folds leak the future into the training set}.}
\end{frame}

% --- Slide: Feature importance -----------------------------------------------
\begin{frame}{Feature Importance \parencite{Molnar2022}}
  \begin{definitionbox}{Impurity-Based Feature Importance}
    For each feature $j$, sum the total reduction in variance across all splits
    where $j$ was used, weighted by the fraction of observations in each node.
    Normalize so the importances sum to 1.
  \end{definitionbox}
  \begin{center}
    {\small
    \begin{tabular}{lc}
      \toprule
      \textbf{Feature (CA\_1 FOODS, weekly)} & \textbf{Importance} \\
      \midrule
      \texttt{units\_lag4}  & 0.62 \\
      \texttt{units\_lag52} & 0.15 \\
      \texttt{units\_lag1}  & 0.13 \\
      \texttt{snap\_days}   & 0.04 \\
      \texttt{avg\_price}   & 0.04 \\
      \bottomrule
    \end{tabular}
    }
  \end{center}
  \vspace{-0.1cm}
  This tells you which variables the tree found most useful for splitting ---
  a fast way to screen predictors.
  \muted{\footnotesize But impurity importance is biased toward
  high-cardinality features. Permutation importance and SHAP give more
  reliable attributions; we return to this in Lecture 5.}
\end{frame}

% --- Slide: Interpreting tree predictions ------------------------------------
\begin{frame}{Interpreting and Communicating Tree Predictions}
  A tree's main advantage is that you can explain \emph{why} a specific
  prediction was made.
  \begin{examplebox}{Explaining a Prediction to a Manager}
    ``Our model forecasts \$61,300 in sales for next month because:
    \begin{enumerate}\small
      \item Last year's same-month sales were above \$38,000 (economy growing)
      \item Last month's sales were above \$52,000 (current momentum strong)
      \item When both conditions hold, the average outcome in our training
            data was \$61,300.''
    \end{enumerate}
  \end{examplebox}
  Produce the path with \texttt{sklearn.tree.plot\_tree(tree)} or
  \texttt{export\_text(tree, feature\_names=[...])}.
  \muted{\footnotesize This is where business buy-in comes from --- a manager
  can check that the tree is splitting on variables they consider sensible.}
\end{frame}

\section{Limitations and Preview: Ensembles}
% =============================================================================

\sectionslide{Limitations and Preview: Ensembles}{%
  A single tree is a weak learner. Combine hundreds to build a strong one.}

% --- Slide: When trees fail ---------------------------------------------------
\begin{frame}{When a Single Tree Falls Short}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Strengths:}
      {\setbeamertemplate{itemize item}{\pos{$\checkmark$}}
      \begin{itemize}\small
        \item Interpretable --- can show the path
        \item No feature scaling needed
        \item Captures nonlinear relationships
        \item Handles missing values well
        \item Fast to train
      \end{itemize}}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Weaknesses:}
      {\setbeamertemplate{itemize item}{\negc{$\times$}}
      \begin{itemize}\small
        \item High variance --- small data changes give a very different tree
        \item Flat predictions within each leaf
        \item Cannot extrapolate beyond the training range
        \item Sensitive to outliers in a leaf
      \end{itemize}}
    \end{column}
  \end{columns}
  \vspace{0.2cm}
  \textbf{The ensemble solution:} average many trees trained on different
  bootstrap samples. Variance falls sharply while feature-importance
  interpretability survives --- typically a 20--40\% RMSE improvement over a
  single tree.
\end{frame}

% --- Slide: Key takeaways ---------------------------------------------------
\begin{frame}{Lecture 4: Key Takeaways}
  \begin{keybox}
    {\small
    \begin{enumerate}
      \item Linear models cannot capture \textbf{interactions} or
            \textbf{threshold effects}. Decision trees can, without
            needing to specify these relationships in advance.

      \item The \textbf{bias--variance tradeoff} means that a
            too-simple tree underfits (high bias) and a too-deep tree
            overfits (high variance). Tune with \texttt{max\_depth}
            and \texttt{min\_samples\_leaf}.

      \item \textbf{CART} greedily minimizes within-region variance at
            each split. Prediction = mean of observations in the leaf.

      \item Decision trees are \textbf{interpretable}: you can trace
            the exact decision path that led to any prediction.

      \item Single trees have \textbf{high variance} --- they change
            dramatically when the data changes slightly.
            Ensembles (Random Forests, XGBoost) solve this.
    \end{enumerate}
    }
  \end{keybox}
  \muted{\itshape Next: Random Forests --- variance reduction through
  bootstrap aggregation and random feature selection.}
\end{frame}

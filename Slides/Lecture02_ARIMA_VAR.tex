\section{Stationarity and Differencing}
% =============================================================================

\sectionslide{Stationarity and Differencing}{%
  Before fitting any model, the series must have a stable mean and variance.}

% --- Slide: Weak stationarity -------------------------------------------------
\begin{frame}{What Is Stationarity?}
  \begin{definitionbox}{Weak (Covariance) Stationarity}
    A time series $\{y_t\}$ is \textbf{weakly stationary} if:
    \begin{enumerate}\small\itemsep1pt
      \item $\E[y_t] = \mu$ \quad (constant mean over time)
      \item $\Var(y_t) = \sigma^2 < \infty$ \quad (constant, bounded variance)
      \item $\Cov(y_t, y_{t-k}) = \gamma_k$ depends only on lag $k$, not on $t$
    \end{enumerate}
  \end{definitionbox}
  \textbf{Why the definition earns its keep:} if the mean drifts, there is no single
  value to forecast. If the variance grows, prediction intervals widen without bound.
  Stationarity is what makes the past informative about the future --- it is the
  assumption every model in this lecture rests on.

  \vspace{0.15cm}
  \textbf{The usual culprits} are trending series --- GDP, prices, sales --- and variance
  that grows with the level. First-differencing removes a stochastic trend; a log
  transform stabilizes variance.

  \vspace{0.1cm}
  \muted{\footnotesize\itshape Rule of thumb: visible trend or growing variance
  $\Rightarrow$ likely non-stationary. Test formally with ADF or KPSS.}
\end{frame}

% --- Slide: Unit root + differencing ------------------------------------------
\begin{frame}{Unit Roots and Differencing}
  \textbf{AR(1):} $y_t = \phi_1\,y_{t-1} + \varepsilon_t$. Everything turns on
  whether $\phi_1$ is below one or exactly one.
  \begin{center}
    {\small
    \begin{tabular}{lll}
      \toprule
      \textbf{Condition} & \textbf{Behavior} & \textbf{Implication} \\
      \midrule
      $|\phi_1| < 1$ & Shock decays, reverts to mean & Stationary AR(1) \\
      $\phi_1 = 1$   & Shock persists \emph{permanently} & Random walk (unit root) \\
      \bottomrule
    \end{tabular}
    }
  \end{center}
  \textbf{First differencing} removes a stochastic trend: $\Delta y_t = y_t - y_{t-1}$.
  So $d=0$ means already stationary, $d=1$ means one first difference, and $D=1$ means a
  seasonal difference, $\Delta_{12}y_t = y_t - y_{t-12}$.

  \vspace{0.15cm}
  \textbf{Testing:} \textbf{ADF} ($H_0$: unit root) and \textbf{KPSS} ($H_0$: stationary)
  test \emph{opposite} nulls --- so run both, and be suspicious when they disagree.
  \textbf{Do not over-difference.} A second difference on an already-stationary series
  induces negative autocorrelation that you then have to model away --- you have
  manufactured the problem you are now solving. Test \emph{after} differencing, and stop at
  $d=1$ once the result is stationary.
\end{frame}

% --- Slide: ACF and PACF for identification -----------------------------------
\begin{frame}{ACF and PACF: Model Identification}
  The \textbf{autocorrelation function}, $\rho_k = \mathrm{Corr}(y_t, y_{t-k})$, measures
  how $y_t$ relates to its own past. The \textbf{partial} autocorrelation measures the same
  thing while controlling for the lags in between --- which is what separates a direct
  effect from one inherited through intermediate lags.

  \vspace{0.1cm}
  Read them together and they identify the model:
  \begin{center}
    {\small
    \begin{tabular}{lll}
      \toprule
      \textbf{Model} & \textbf{ACF} & \textbf{PACF} \\
      \midrule
      AR($p$)   & Tails off geometrically & \pos{Cuts off} after lag $p$ \\
      MA($q$)   & \pos{Cuts off} after lag $q$ & Tails off geometrically \\
      ARMA      & Tails off & Tails off \\
      RW / I(1) & Decays slowly & Big spike at lag 1 \\
      \bottomrule
    \end{tabular}
    }
  \end{center}
  \vspace{0.1cm}
  \textbf{Always unit-root test and difference \emph{before} reading either plot.} The ACF
  of a non-stationary series decays slowly no matter what the underlying process is, so it
  tells you nothing about $p$ or $q$ --- only that you have not differenced yet.
\end{frame}

\section{ARIMA Models}
% =============================================================================

\sectionslide{ARIMA Models}{%
  Combine differencing with AR and MA components into one unified model.}

% --- Slide: ARIMA definition --------------------------------------------------
\begin{frame}{ARIMA($p$,$d$,$q$)}
  \begin{definitionbox}{ARIMA($p$,$d$,$q$) Model}
    {\small Apply an ARMA($p$,$q$) to the $d$-times differenced series:
    $y_t = c + \sum_{i=1}^{p}\phi_i y_{t-i} + \varepsilon_t + \sum_{j=1}^{q}\theta_j\varepsilon_{t-j}$}
  \end{definitionbox}
  $p$ is the number of AR lags, $d$ the number of differences, $q$ the number of MA lags.
  Several familiar models are just corners of this space: \textbf{ARIMA(0,1,0)} is a random
  walk, \textbf{ARIMA(1,0,0)} is an AR(1), and \textbf{ARIMA(0,1,1)} is algebraically
  equivalent to simple exponential smoothing from Lecture 1.

  \vspace{0.15cm}
  \textbf{A forecast by hand.} Take ARIMA(1,1,0) with $\hat{\phi}_1 = 0.5$. If
  $y_{100} = 120$ and $y_{99} = 118$, then $\Delta y_{100} = 2$, and
  \[
    \hat{y}_{101|100} \;=\; 120 + 0.5 \times 2 \;=\; \mathbf{121}.
  \]
  \vspace{-0.2cm}
  \textbf{In Python:} \texttt{pmdarima.auto\_arima()} searches orders automatically;
  \texttt{statsmodels.SARIMAX} fits any order you specify.
\end{frame}

% --- Slide: SARIMA ------------------------------------------------------------
\begin{frame}{SARIMA: Adding Seasonal Terms}
  For monthly or quarterly data, autocorrelation often appears at lags $m, 2m, 3m, \ldots$
  SARIMA($p$,$d$,$q$)($P$,$D$,$Q$)[$m$] adds seasonal AR and MA polynomials alongside the
  non-seasonal ones.

  \vspace{0.15cm}
  The useful mental model is \textbf{two ARIMA models running at different time scales}:
  the non-seasonal part $(p,d,q)$ captures week-to-week dynamics, while the seasonal part
  $(P,D,Q)[m]$ captures the year-over-year pattern at lag $m$.

  \vspace{0.15cm}
  \textbf{Practical workflow:}
  \begin{enumerate}\small\itemsep1pt
    \item ADF / KPSS $\to$ choose $d$
    \item Seasonal unit root test $\to$ choose $D$
    \item ACF / PACF on the differenced series $\to$ guide $p, q, P, Q$
    \item Minimize AIC, then check the residual ACF with a Ljung-Box test
  \end{enumerate}
  \vspace{0.1cm}
  \muted{\small For monthly retail data with trend and December seasonality,
  SARIMA(1,1,1)(0,1,1)[12] is a strong starting point --- and a good default to beat
  before trying anything more elaborate.}
\end{frame}

\section{VAR Models}
% =============================================================================

\sectionslide{VAR Models}{%
  When two series influence each other, model them together symmetrically.}

% --- Slide: Why multivariate --------------------------------------------------
\begin{frame}{Why Use Multiple Series?}
  Every model so far --- ETS, ARIMA --- uses only $y_t$'s own history. But business
  variables are linked: consumer sentiment today moves retail spending next month,
  interest rates move housing starts within two quarters, advertising spend moves sales
  over the following weeks.

  \vspace{0.1cm}
  The principle underneath all of it: \textbf{if $x_t$ carries information about $y_{t+h}$
  that is not already in $y_t$'s own history, including $x_t$ reduces forecast error.}
  \textbf{Granger causality} \parencite{Granger1969} is the formal test of exactly that
  claim, and we get to it in two slides.

  \vspace{0.15cm}
  That leaves a modeling choice, and it turns on whether the influence runs one way or both:
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{VAR --- symmetric}
      \begin{itemize}\small\itemsep0pt
        \item Every variable predicts every other
        \item Use when influence runs both ways
        \item No variable is privileged
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{ARIMAX --- one direction}
      \begin{itemize}\small\itemsep0pt
        \item $x_t$ drives $y_t$, not the reverse
        \item Use when the causal story is clear
        \item Needs a forecast of $x_t$ itself
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% --- Slide: VAR(1) definition -------------------------------------------------
\begin{frame}{The VAR($p$) Model}
  For two stationary series $y_{1t}$ and $y_{2t}$, each equation regresses on the lags of
  \emph{both}:
  \[
    \begin{cases}
      y_{1t} = c_1 + a_{11}\,y_{1,t-1} + a_{12}\,y_{2,t-1} + \varepsilon_{1t}\\[2pt]
      y_{2t} = c_2 + a_{21}\,y_{1,t-1} + a_{22}\,y_{2,t-1} + \varepsilon_{2t}
    \end{cases}
  \]
  The cross terms are where the value is: $a_{12}$ is how much $y_{2,t-1}$ predicts $y_{1t}$,
  which is precisely what a univariate AR throws away. If $a_{12} = a_{21} = 0$, the system
  collapses back into two separate AR models and you have learned that too.
  Each equation is estimated by OLS; the order $p$ is chosen by BIC.
  \begin{warningbox}
    \textbf{Parameter explosion.} A VAR($p$) with $k$ variables has $k^2 p$ coefficients.
    At $k=5$ and $p=4$ that is \textbf{100 coefficients} --- typically more than a monthly
    business series has observations. Keep $k \leq 4$ unless you are using a LASSO-VAR.
  \end{warningbox}
  \muted{\footnotesize \textbf{Python:}
  \texttt{statsmodels.tsa.api.VAR}; \texttt{var.fit(maxlags=p, ic='bic')} selects the order.}
\end{frame}

% --- Slide: Granger causality -------------------------------------------------
\begin{frame}{Granger Causality Test}
  \begin{definitionbox}{Granger Causality}
    $x_t$ \textbf{Granger-causes} $y_t$ if past values of $x_t$ help predict $y_t$
    \emph{beyond what $y_t$'s own past already explains}.
  \end{definitionbox}
  \textbf{The test:} fit the $y$-equation with and without the $x$-lags, then run an
  $F$-test of $H_0$: all $x$-lag coefficients are zero. Rejecting means $x_t$
  Granger-causes $y_t$.

  \vspace{0.1cm}
  \textbf{A concrete question.} Does unemployment (\texttt{UNRATE}) Granger-cause retail
  sales (\texttt{RSXFS})? \textbf{We run exactly this in the lab} --- and the answer is
  instructive: it fires \emph{both} ways, and which direction looks stronger depends on
  whether COVID is in the sample.

  \vspace{0.15cm}
  \textbf{Granger causality is not structural causality.} It tests \emph{predictive
  content}, not whether $x$ brings $y$ about --- a rooster reliably predicts sunrise.
  Report it as ``predictive of,'' never ``causes.''

  \muted{\footnotesize\itshape Impulse response functions trace how a shock propagates
  through the system.}
\end{frame}

% --- Slide: ARIMAX -----------------------------------------------------------
\begin{frame}{ARIMAX: Exogenous Variables with ARIMA Errors}
  When one variable clearly \emph{drives} another rather than the relationship being
  symmetric, use ARIMAX.
  \begin{definitionbox}{ARIMAX Model}
    \[
      y_t = \mathbf{x}_t'\boldsymbol{\beta} + \eta_t,
      \qquad \eta_t \sim \mathrm{ARIMA}(p,d,q)
    \]
    $\mathbf{x}_t$ are exogenous regressors; $\eta_t$ absorbs whatever autocorrelation is
    left in the residuals.
  \end{definitionbox}
  \textbf{Use it when} the causal direction is clear (advertising $\to$ sales), you already
  have $x_t$ or can forecast it, and adding $x_t$ actually reduces cross-validated forecast
  error --- check that last one rather than assuming it.

  \vspace{0.15cm}
  \textbf{The catch that surprises people:} to forecast $y_{t+h}$ you need
  $x_{t+h}$ --- a forecast of your own predictor. If that is unavailable, ARIMAX buys you
  nothing at forecast time and you should use a VAR instead, which generates its own.

  \vspace{0.1cm}
  \muted{\footnotesize \textbf{Python:} \texttt{SARIMAX(y, exog=X, order=(p,d,q)).fit()}}
\end{frame}

\section{Key Takeaways and Roadmap}
% =============================================================================

\sectionslide{Key Takeaways and Roadmap}{%
  Stationarity, ARIMA, VAR, and ARIMAX form the classical time series toolkit.}

% --- Slide: Key Takeaways -----------------------------------------------
\begin{frame}{Key Takeaways}
  \begin{keybox}
    {\small
    \begin{enumerate}
      \item \textbf{Stationarity} is required before fitting time series models.
            Test with ADF ($H_0$: unit root) and KPSS ($H_0$: stationary).
            First-differencing removes a stochastic trend.

      \item \textbf{ACF} identifies MA order; \textbf{PACF} identifies AR order.
            Always inspect after differencing, not before.

      \item \textbf{ARIMA($p$,$d$,$q$)} combines differencing with AR/MA.
            SARIMA adds seasonal polynomials for periodic autocorrelation.
            Use \texttt{auto\_arima} for automatic selection.

      \item \textbf{VAR} models all variables symmetrically and captures
            cross-variable dynamics. Keep $k \leq 4$.
            Use \textbf{Granger causality} to test predictive content.

      \item \textbf{ARIMAX} adds exogenous predictors to ARIMA.
            Use when causal direction is clear and $x_t$ is forecastable.
    \end{enumerate}
    }
  \end{keybox}
  \muted{\itshape Next: Generalized Additive Models (GAMs) ---
  flexible, interpretable nonlinear forecasting with Prophet and pyGAM.}
\end{frame}
